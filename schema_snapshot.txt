analytics_attendance(id uuid pk, user_id uuid fk, franchise_id uuid fk, warehouse_id uuid fk, date date, check_in_at timestamptz, check_out_at timestamptz, status text, working_hours numeric, face_verified bool, is_within_geofence bool, created_at timestamptz)  -- status is one of: present, absent, late, half_day, leave (plain text with a CHECK constraint, not a native enum - ILIKE works, but exact = is preferred). No row is ever created for a day someone didn't check in - a missed day is a MISSING row, not a row with status='absent'. Never compute attendance % as COUNT(attendance rows)/COUNT(attendance rows) - that is always 100%. Use this pattern instead, which generates every expected working day and LEFT JOINs actual attendance onto it so a missing day correctly counts against the person:

  WITH expected_days AS (
    SELECT u.id AS user_id, u.full_name, d::date AS work_date
    FROM analytics_users u
    CROSS JOIN generate_series(CURRENT_DATE - INTERVAL '1 month', CURRENT_DATE - INTERVAL '1 day', '1 day'::interval) AS d
    WHERE u.is_active = true AND EXTRACT(DOW FROM d) != 0  -- excludes Sundays; adjust to the actual work week if asked
  )
  SELECT ed.user_id, ed.full_name, COUNT(a.id) * 100.0 / COUNT(ed.work_date) AS attendance_percentage
  FROM expected_days ed
  LEFT JOIN analytics_attendance a ON a.user_id = ed.user_id AND a.date = ed.work_date
  GROUP BY ed.user_id, ed.full_name  -- group by user_id, not full_name alone: two different active
  -- employees can share the same full_name, and grouping by name alone silently merges their
  -- attendance into one blended, meaningless row.
  ORDER BY attendance_percentage ASC;

chat_conversations(id uuid pk, user_a uuid fk, user_b uuid fk, last_read_a timestamptz, last_read_b timestamptz, last_message_at timestamptz, last_message_preview text, created_at timestamptz, updated_at timestamptz)  -- one row per 1-to-1 internal staff conversation, not per direction. user_a is always whichever of the two user ids sorts smaller (user_a < user_b is a DB constraint) - it is NOT "who started the chat" and carries no meaning beyond sort order, so never describe user_a as the sender/initiator or user_b as the recipient. To name both participants, JOIN analytics_users twice (once on user_a, once on user_b). last_message_preview is a denormalized snippet kept in sync by a trigger purely for an inbox-list UI - for the actual message text, query chat_messages instead of trusting this field as complete.
chat_messages(id uuid pk, conversation_id uuid fk, sender_id uuid fk, recipient_id uuid fk, body text, attachment_path text, attachment_name text, attachment_mime text, attachment_size bigint, created_at timestamptz)  -- 1-to-1 internal staff direct messages. Permanent history: there is no edit or delete in this product, so every row that was ever sent still exists exactly as sent. body can be null when the message is attachment-only; every row has a non-empty body and/or an attachment, never neither. attachment_path is a private Supabase Storage path, not a usable URL - don't present it as a link or guess at file contents from it; attachment_name/attachment_mime/attachment_size are the only attachment detail available. recipient_id is denormalized (it's always just "the other user in conversation_id") to support a realtime subscription filter - resolve both sender_id and recipient_id to analytics_users.full_name as usual, same as any other *_id column. "Unread" is not a column on either table - a given message counts as unread for a user when its created_at is later than that user's own last_read_a/last_read_b (whichever side of chat_conversations they are) on the parent conversation row.
